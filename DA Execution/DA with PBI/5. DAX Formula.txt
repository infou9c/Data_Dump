Calculate overall quantity sold

Calculate gross sales as Quantity multiplied by UnitPrice row by row
	Gross Sales = SUMX(Sales, Sales[Quantity] * Sales[UnitPrice])
		- Looping

Count number of sale transaction
	Sales Count = COUNTROWS(Sales)
	
	Additional DAX:
	COUNT(Table[ColumnName]) => Used for counting. Skips blank rows
	COUNTBLANK(Table[ColumnName]) => Counts blank rows
	COUNTA(Table[ColumnName]) => Used to count where datatype is boolean.
	DISTINCTCOUNT(Table[ColumnName]) => Count distinct values for the input column.
	COUNTX(Table, FilterExpression) => Count with a condition.
		- COUNTX(Table, if(Table[ColumnName] > 5, 1, BLANK()))
			- It will count all "1" and return total count.
		- COUNTX(FILTER(Table, Table[ColumnName] > 5), Table[ColumnName]) 
	COUNTAX(Table, FilterExpression) => Count with a condition for boolean datatypes.
	COUNTROWS(Table, FilterExpression) => Count with a filter context.
		- COUNTROWS(FILTER(Table, Table[ColumnName] > 5))

Calculate sales only for Product = Electronics 
	Electronics Sales = CALCULATE([Total Sales], Product[Category] = "Electronics")

Calculate sales for West region
	West Sales = CALCULATE([Total Sales], Customer[Region] = "West")
	West Sales = CALCULATE([Total Sales], FILTER(Customer, Customer[Region] = "West"))
	
Calculate total sales where quantity is greater than 3
	High Quantity Sales = CALCULATE([Total Sales], FILTER(Sales, Sales[Quantity] > 3))
	
Calculate weekend-day sales excluding weekdays
	Weekend Sales = CALCULATE([Total Sales], 'Date'[IsWeekend] = TRUE())

Count high-value customers whose sales exceed ₹100,000
	High Value Customers = COUNTROWS(FILTER(VALUES(Customer[CustomerID]), CALCULATE([Total Sales]) > 100000))
	
Count customers with purchases in both 2024 and 2025
	Customers In 24-25 = VAR C2024 = CALCULATETABLE(VALUES(Sales[CustomerID]), 'Date'[Year] = 2024) VAR C2025 = CALCULATETABLE(VALUES(Sales[CustomerID]), 'Date'[Year] = 2025) RETURN COUNTROWS(INTERSECT(C2024, C2025))

	Quantity MTD = TOTALMTD([Total Quantity], 'Date'[Date])
		- Total of Month 1 Day 1 = Day 1
		- Total of Month 1 Day 2 = Day 1 + Day 2
		- Total of Month 1 Day 3 = Day 1 + Day 2 + Day 3
		- Total of Month 1 Day 30 = Day 1 + ... + Day 29
		- Total of Month 2 Day 1 = Day 1
		- Total of Month 2 Day 2 = Day 1 + Day 2
	Quantity QTD = TOTALQTD([Total Quantity], 'Date'[Date])
		- Total of Quarter 1 Month 1 = Month 1
		- Total of Quarter 1 Month 2 = Month 1 + Month 2
		- Total of Quarter 1 Month 3 = Month 1 +Month 2 + Month 3
		- Total of Quarter 2 Month 1 = Month 1
		- Total of Quarter 2 Month 2 = Month 1 + Month 2
	Quantity YTD = TOTALYTD([Total Quantity], 'Date'[Date])
		- Total of Year 1 Quarter 1 = Quarter 1
		- Total of Year 1 Quarter 2 = Quarter 1 + Quarter 2
		- Total of Year 1 Quarter 3 = Quarter 1 +Quarter 2 + Quarter 3
		- Total of Year 1 Quarter 4 = Quarter 1 +Quarter 2 + Quarter 3 + Quarter 4
		- Total of Year 2 Quarter 1 = Quarter 1
		- Total of Year 2 Quarter 2 = Quarter 1 + Quarter 2
